% FZU-inspired header: logo and personal details share one vertically centred row.
% Normal document flow lets the header grow safely when optional fields are used.
\newcommand{\MakeLetterHeader}{%
  \begingroup
  \setlength{\parskip}{0pt}%
  \renewcommand{\tabularxcolumn}[1]{m{##1}}%
  \noindent
  \begin{tabularx}{\textwidth}{%
    @{}m{0.50\textwidth}@{\hspace{5mm}}%
    !{\color{RuleInk}\vrule width 0.8pt}%
    @{\hspace{3mm}}>{\raggedright\arraybackslash}X@{}%
  }

    \includegraphics[width=\UQLogoWidth]{\UQLogoFile}
    &
    % Applicant Name
    {\fontfamily{phv}\fontseries{b}
      \fontsize{15}{17}\selectfont
      \color{UQPurple}%
      \ApplicantName\par}%

    \vspace{0mm}%

    % Personal Details
    {\fontfamily{phv}\fontsize{9}{14}\selectfont
      \color{MutedInk}%

      % Position + Institution
      \noindent\hspace*{0.3mm}%
      \ifdefempty{\ApplicantRole}{}{%
        {\bfseries\color{BodyInk}\ApplicantRole}%
      }%
      \ifdefempty{\ApplicantAffiliation}{}{%
        \ifdefempty{\ApplicantRole}{}{\enspace|\enspace}%
        \ApplicantAffiliation
      }%
      \par

      % Address
      \ifdefempty{\ApplicantAddress}{}{%
        \ApplicantAddress\par
      }%

      % Location
      \ifdefempty{\ApplicantLocation}{}{%
        \ContactIcon{\faIcon{map-marker-alt}}%
        \ApplicantLocation\par
      }%

      % Phone
      \ifdefempty{\ApplicantPhone}{}{%
        \ContactIcon{\faPhone}%
        \ApplicantPhone\par
      }%

      % Email
      \href{mailto:\ApplicantEmail}{%
        \ContactIcon{\faEnvelope}\ApplicantEmail
      }\par

      % Website
      \ifdefempty{\ApplicantWebsite}{}{%
        \href{\ApplicantWebsite}{%
          \ContactIcon{\faGlobe}\ApplicantWebsiteLabel
        }\par
      }%
    }%

    \\
  \end{tabularx}\par
  \vspace{4mm}%
  {\color{UQPurple}\hrule height 0.6pt}%
  \endgroup
  \vspace{5mm}%
}
